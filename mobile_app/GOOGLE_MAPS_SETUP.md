# Google Maps API Key Configuration
# 
# To enable map functionality in the mobile app, you need to add your Google Maps API key:
#
# Android: Replace "YOUR_GOOGLE_MAPS_API_KEY" in:
#   android/app/build.gradle - manifestPlaceholders
#   android/app/src/main/AndroidManifest.xml
#
# iOS: Replace "YOUR_GOOGLE_MAPS_API_KEY" in:
#   ios/Runner/AppDelegate.swift - GMSServices.provideAPIKey()
#   ios/Runner/Info.plist - GMSApiKey
#
# Get your API key from: https://console.cloud.google.com/google/maps-apis
